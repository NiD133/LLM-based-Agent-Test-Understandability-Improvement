                 /**
                  * Verifies that, when the variable prefix is set to the same character as the
                  * default escape character ('$'), {@link StrSubstitutor#replace(Object)} leaves
                  * the input unchanged because every "prefix" is treated as an escaped literal.
                  */
                 @Test(timeout = 4000)
                 public void replaceLeavesTextUnchangedWhenPrefixEqualsEscapeChar() throws Throwable {
                     // The substitutor has no variables to resolve (empty map).
                     Map<String, String> emptyVariables = new HashMap<String, String>();
                     StrSubstitutor substitutor = new StrSubstitutor(emptyVariables, PLAIN_TEXT, PLAIN_TEXT);
             
                     // Make the variable prefix '$', which is also the default escape character.
                     substitutor.setVariablePrefix('$');
             
                     // Build "PLAIN_TEXT" + "$$" + three NUL chars, then append the whole thing to itself.
                     char[] prefixAndPadding = new char[5];
                     prefixAndPadding[0] = '$';
                     prefixAndPadding[1] = '$';
                     // prefixAndPadding[2..4] remain ' '
             
                     StringBuilder source = new StringBuilder((CharSequence) PLAIN_TEXT);
                     source.append(prefixAndPadding);
                     source.append((CharSequence) source);
             
                     String result = substitutor.replace((Object) source);
             
                     // The default escape character is unchanged by setting the prefix.
                     assertEquals('$', substitutor.getEscapeChar());
                     // No substitution happens; the result equals the (doubled) source text.
                     String expected = PLAIN_TEXT + "$$   " + PLAIN_TEXT + "$$   ";
                     assertEquals(expected, result);
                 }
